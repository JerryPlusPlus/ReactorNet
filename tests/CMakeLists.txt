# Compile every shipped header in its own translation unit.
file(GLOB_RECURSE public_headers CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/ReactorNet/*.h")
set(header_sources)
foreach(header IN LISTS public_headers)
  if(NOT ZLIB_FOUND AND header MATCHES "(GzipFile|ZlibStream)\\.h$")
    continue()
  endif()
  file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}" "${header}")
  string(REPLACE "/" "_" source_name "${relative}")
  set(source "${CMAKE_CURRENT_BINARY_DIR}/headers/${source_name}.cc")
  file(GENERATE OUTPUT "${source}" CONTENT "#include \"${relative}\"\n")
  list(APPEND header_sources "${source}")
endforeach()
add_library(reactornet_header_checks OBJECT ${header_sources})
target_link_libraries(reactornet_header_checks PRIVATE ReactorNet::net)
target_compile_options(reactornet_header_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ZLIB_FOUND)
  target_link_libraries(reactornet_header_checks PRIVATE ZLIB::ZLIB)
endif()
