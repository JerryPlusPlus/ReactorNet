cmake_minimum_required(VERSION 3.16)
project(ReactorNet VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "ReactorNet requires Linux (Windows users can build in WSL).")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(REACTORNET_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(REACTORNET_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
include(GNUInstallDirs)
add_subdirectory(ReactorNet/base)
add_subdirectory(ReactorNet/log)
add_subdirectory(ReactorNet/net)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_subdirectory(tests)
endif()
option(REACTORNET_BUILD_BENCHMARKS "Build echo server and TCP benchmark client" ON)
if(REACTORNET_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()
install(DIRECTORY ReactorNet/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/ReactorNet
        FILES_MATCHING PATTERN "*.h" PATTERN "boilerplate.h" EXCLUDE)
